<script>
const DB_URL = "https://TU-PROYECTO-default-rtdb.firebaseio.com"; // reemplaza por tu URL
let registros = [];

const esc = s => String(s ?? "").replace(/[&<>"']/g,
  c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));

async function api(metodo, ruta, cuerpo, params = "") {
  const r = await fetch(`${DB_URL}/${ruta}.json${params}`, {
    method: metodo,
    body: cuerpo !== undefined ? JSON.stringify(cuerpo) : undefined
  });
  const datos = await r.json();
  if (datos && datos.error) throw new Error(datos.error);
  return datos;
}

function fechaHora(ts) {
  const d = new Date(ts);
  const p = n => String(n).padStart(2, "0");
  return {
    fecha: d.toLocaleDateString("es-CO"),
    hora: d.toLocaleTimeString("es-CO"),
    iso: `${d.getFullYear()}-${p(d.getMonth() + 1)}-${p(d.getDate())}`
  };
}

let datos = {};          // copia local de la base de datos
let pendienteRender = false;

function render() {
  pendienteRender = false;
  registros = Object.values(datos.registros || {}).sort((x, y) => y.ts - x.ts).slice(0, 1000);
  document.getElementById("cuenta").textContent =
    Object.values(datos.abierto || {}).filter(v => v === true).length;
  pintarPendientes(datos.pendientes || {});
  pintarEstudiantes(datos.estudiantes || {});
  llenarSelectManual(datos.estudiantes || {});
  pintarRegistros();
}
function cargar() {
  if (!pendienteRender) { pendienteRender = true; requestAnimationFrame(render); }
}

// Aplica un cambio {path, data} de Firebase a la copia local
function aplicar(path, data, mezclar) {
  const partes = path.split("/").filter(Boolean);
  if (!partes.length) {
    datos = mezclar ? Object.assign(datos, data || {}) : (data || {});
    return;
  }
  let nodo = datos;
  for (let i = 0; i < partes.length - 1; i++) {
    if (typeof nodo[partes[i]] !== "object" || nodo[partes[i]] === null) nodo[partes[i]] = {};
    nodo = nodo[partes[i]];
  }
  const k = partes[partes.length - 1];
  if (mezclar && data && typeof data === "object") {
    nodo[k] = Object.assign(nodo[k] || {}, data);
  } else if (data === null) {
    delete nodo[k];
  } else {
    nodo[k] = data;
  }
}

function estado(texto, ok) {
  const el = document.getElementById("estado");
  el.textContent = (ok ? "● " : "○ ") + texto;
  el.style.color = ok ? "#16a34a" : "#dc2626";
}

function conectar() {
  const es = new EventSource(`${DB_URL}/.json`);
  es.addEventListener("open", () => estado("En vivo", true));
  es.addEventListener("put", e => { const m = JSON.parse(e.data); aplicar(m.path, m.data, false); cargar(); });
  es.addEventListener("patch", e => { const m = JSON.parse(e.data); aplicar(m.path, m.data, true); cargar(); });
  es.addEventListener("cancel", () => estado("Sin permiso (revisa las reglas de Firebase)", false));
  es.addEventListener("auth_revoked", () => estado("Sesión revocada", false));
  es.onerror = () => estado("Reconectando...", false);
}

function pintarPendientes(pend) {
  const cont = document.getElementById("pendientes");
  // Conservar lo que el usuario esté escribiendo
  const guardado = {};
  cont.querySelectorAll("input").forEach(i => guardado[i.id] = i.value);
  const enfocado = document.activeElement && cont.contains(document.activeElement) ? document.activeElement.id : null;
  const uids = Object.keys(pend);
  if (!uids.length) { cont.innerHTML = '<p class="vacio">Sin tarjetas pendientes.</p>'; return; }
  cont.innerHTML = uids.map(uid => `
    <div class="pend">
      <code>${esc(uid)}</code>
      <input id="n-${esc(uid)}" placeholder="Nombre completo">
      <input id="d-${esc(uid)}" placeholder="Documento">
      <button onclick="asignar('${esc(uid)}')">Asignar</button>
      <button class="sec" onclick="descartar('${esc(uid)}')">Descartar</button>
    </div>`).join("");
  cont.querySelectorAll("input").forEach(i => { if (guardado[i.id]) i.value = guardado[i.id]; });
  if (enfocado) { const f = document.getElementById(enfocado); if (f) f.focus(); }
}

function llenarSelectManual(est) {
  const sel = document.getElementById("manual-est");
  const actual = sel.value;
  const lista = Object.entries(est).sort((a, b) => String(a[1].nombre).localeCompare(String(b[1].nombre)));
  sel.innerHTML = '<option value="">— elige estudiante —</option>' + lista.map(([uid, e]) =>
    `<option value="${esc(uid)}">${esc(e.nombre)} (${esc(e.documento)})</option>`).join("");
  if (actual && est[actual]) sel.value = actual;
}

async function registrarManual(tipo) {
  const uid = document.getElementById("manual-est").value;
  const msg = document.getElementById("manual-msg");
  if (!uid) { msg.textContent = "Elige un estudiante."; return; }
  const e = (datos.estudiantes || {})[uid];
  if (!e) { msg.textContent = "Ese estudiante ya no existe."; return; }
  const dentro = (datos.abierto || {})[uid] === true;
  if (tipo === "entrada" && dentro) { msg.textContent = e.nombre + " ya figura dentro."; return; }
  if (tipo === "salida" && !dentro) { msg.textContent = e.nombre + " ya figura fuera."; return; }
  try {
    await api("POST", "registros", { uid, nombre: e.nombre, documento: e.documento, tipo, manual: true, ts: { ".sv": "timestamp" } });
    await api("PUT", "abierto/" + uid, tipo === "entrada");
    msg.textContent = "✔ " + tipo[0].toUpperCase() + tipo.slice(1) + " registrada para " + e.nombre + ".";
  } catch (err) {
    msg.textContent = "No se pudo registrar: " + err.message;
  }
  cargar();
}

function pintarEstudiantes(est) {
  document.getElementById("estudiantes").innerHTML = Object.entries(est).map(([uid, e]) => `
    <tr><td>${esc(e.nombre)}</td><td>${esc(e.documento)}</td><td><code>${esc(uid)}</code></td>
    <td><button class="sec" onclick="eliminarEst('${esc(uid)}')">Eliminar</button></td></tr>`).join("")
    || '<tr><td colspan="4" class="vacio">Aún no hay estudiantes.</td></tr>';
}

function filtrados() {
  const q = document.getElementById("buscar").value.toLowerCase();
  const f = document.getElementById("fecha").value;
  const t = document.getElementById("tipo").value;
  return registros.filter(r =>
    (!q || (r.nombre || "").toLowerCase().includes(q) || String(r.documento).includes(q)) &&
    (!f || fechaHora(r.ts).iso === f) &&
    (!t || r.tipo === t));
}

function pintarRegistros() {
  const lista = filtrados();
  document.getElementById("filas").innerHTML = lista.map(r => {
    const fh = fechaHora(r.ts);
    return `<tr><td>${fh.fecha}</td><td>${fh.hora}</td><td>${esc(r.nombre)}</td>
            <td>${esc(r.documento)}</td><td class="${esc(r.tipo)}">${esc(r.tipo)}</td></tr>`;
  }).join("") || '<tr><td colspan="5" class="vacio">Sin registros.</td></tr>';
}

async function asignar(uid) {
  const nombre = document.getElementById("n-" + uid).value.trim();
  const documento = document.getElementById("d-" + uid).value.trim();
  if (!nombre || !documento) return alert("Completa nombre y documento");
  await api("PUT", "estudiantes/" + uid, { nombre, documento });
  await api("DELETE", "pendientes/" + uid);
  cargar();
}
async function descartar(uid) { await api("DELETE", "pendientes/" + uid); cargar(); }
async function eliminarEst(uid) {
  if (!confirm("¿Eliminar este estudiante?")) return;
  await api("DELETE", "estudiantes/" + uid);
  await api("DELETE", "abierto/" + uid);
  cargar();
}

function limpiarFiltros() {
  document.getElementById("buscar").value = "";
  document.getElementById("fecha").value = "";
  document.getElementById("tipo").value = "";
  pintarRegistros();
}

function exportarCSV() {
  const celda = v => `"${String(v ?? "").replace(/"/g, '""')}"`;
  const filas = [["Fecha","Hora","Nombre","Documento","Tipo"]].concat(
    filtrados().map(r => { const fh = fechaHora(r.ts); return [fh.fecha, fh.hora, r.nombre, r.documento, r.tipo]; }));
  const csv = "﻿" + filas.map(f => f.map(celda).join(";")).join("\r\n");
  const a = document.createElement("a");
  a.href = URL.createObjectURL(new Blob([csv], { type: "text/csv;charset=utf-8" }));
  a.download = "registros_acceso.csv";
  a.click();
}

["buscar", "fecha", "tipo"].forEach(id =>
  document.getElementById(id).addEventListener("input", pintarRegistros));

conectar();
</script>
